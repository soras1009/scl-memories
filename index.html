const SHEET_ID = '1B6M1DNSn53sx6CYnv6S0ky4LdgSEuF8kklgzDQA4QqQ'; const FOLDER_ID = '1EA3NVFerBYG-PfknAPybw43jhFj9MUkr'; const ADMIN_PW = 'scl2024!'; // ← 관리자 비밀번호 여기서 변경 가능 function doGet(e) { const action = e.parameter.action; try { if (action === 'getPosts') return res(getPosts()); if (action === 'getComments') return res(getComments(e.parameter.postId)); if (action === 'adminCheck') return res(adminCheck(e.parameter.pw)); return res({ error: 'Unknown action' }); } catch(err) { return res({ error: err.message }); } } function doPost(e) { const data = JSON.parse(e.postData.contents); const action = data.action; try { if (action === 'addPost') return res(addPost(data)); if (action === 'addComment') return res(addComment(data)); if (action === 'toggleLike') return res(toggleLike(data)); if (action === 'deletePost') return res(deletePost(data)); if (action === 'editPost') return res(editPost(data)); if (action === 'deleteComment') return res(deleteComment(data)); if (action === 'adminCheck') return res(adminCheck(data.pw)); return res({ error: 'Unknown action' }); } catch(err) { return res({ error: err.message }); } } function res(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function getSheet(name) { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); if (name === 'posts') sheet.appendRow(['id','name','company','team','story','imageUrl','likes','ts']); if (name === 'comments') sheet.appendRow(['id','postId','name','company','team','text','ts']); } return sheet; } function getPosts() { const sheet = getSheet('posts'); const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return []; const headers = rows[0]; return rows.slice(1).map(r => { const obj = {}; headers.forEach((h, i) => obj[h] = r[i]); obj.likes = Number(obj.likes) || 0; obj.ts = Number(obj.ts) || 0; return obj; }).reverse(); } function addPost(data) { const sheet = getSheet('posts'); const id = Utilities.getUuid(); let imageUrl = ''; if (data.imageBase64 && data.imageBase64.length > 0) { const folder = DriveApp.getFolderById(FOLDER_ID); const blob = Utilities.newBlob( Utilities.base64Decode(data.imageBase64.split(',')[1]), data.imageMime || 'image/jpeg', id + '.jpg' ); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); imageUrl = 'https://drive.google.com/uc?export=view&id=' + file.getId(); } sheet.appendRow([id, data.name, data.company, data.team, data.story, imageUrl, 0, Date.now()]); return { success: true, id }; } function editPost(data) { const sheet = getSheet('posts'); const rows = sheet.getDataRange().getValues(); const headers = rows[0]; const idIdx = headers.indexOf('id'); const nameIdx = headers.indexOf('name'); const teamIdx = headers.indexOf('team'); const storyIdx= headers.indexOf('story'); for (let i = 1; i < rows.length; i++) { if (String(rows[i][idIdx]) === String(data.postId)) { if (!data.isAdmin && (rows[i][nameIdx] !== data.name || rows[i][teamIdx] !== data.team)) { return { error: '본인만 수정할 수 있어요.' }; } sheet.getRange(i + 1, storyIdx + 1).setValue(data.story); return { success: true }; } } return { error: '게시글을 찾을 수 없어요.' }; } function deletePost(data) { const sheet = getSheet('posts'); const rows = sheet.getDataRange().getValues(); const headers = rows[0]; const idIdx = headers.indexOf('id'); const nameIdx = headers.indexOf('name'); const teamIdx = headers.indexOf('team'); const imgIdx = headers.indexOf('imageUrl'); for (let i = 1; i < rows.length; i++) { if (String(rows[i][idIdx]) === String(data.postId)) { if (!data.isAdmin && (rows[i][nameIdx] !== data.name || rows[i][teamIdx] !== data.team)) { return { error: '본인만 삭제할 수 있어요.' }; } try { const imgUrl = rows[i][imgIdx]; if (imgUrl && imgUrl.includes('id=')) { DriveApp.getFileById(imgUrl.split('id=')[1]).setTrashed(true); } } catch(e) {} sheet.deleteRow(i + 1); deleteCommentsByPostId(String(data.postId)); return { success: true }; } } return { error: '게시글을 찾을 수 없어요.' }; } function getComments(postId) { const sheet = getSheet('comments'); const rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return []; const headers = rows[0]; return rows.slice(1) .map(r => { const obj = {}; headers.forEach((h, i) => obj[h] = r[i]); return obj; }) .filter(c => String(c.postId) === String(postId)); } function addComment(data) { const sheet = getSheet('comments'); const id = Utilities.getUuid(); sheet.appendRow([id, data.postId, data.name, data.company, data.team, data.text, Date.now()]); return { success: true, id }; } function deleteComment(data) { const sheet = getSheet('comments'); const rows = sheet.getDataRange().getValues(); const headers = rows[0]; const idIdx = headers.indexOf('id'); const nameIdx = headers.indexOf('name'); const teamIdx = headers.indexOf('team'); for (let i = 1; i < rows.length; i++) { if (String(rows[i][idIdx]) === String(data.commentId)) { if (!data.isAdmin && (rows[i][nameIdx] !== data.name || rows[i][teamIdx] !== data.team)) { return { error: '본인만 삭제할 수 있어요.' }; } sheet.deleteRow(i + 1); return { success: true }; } } return { error: '댓글을 찾을 수 없어요.' }; } function deleteCommentsByPostId(postId) { const sheet = getSheet('comments'); const rows = sheet.getDataRange().getValues(); const headers = rows[0]; const pidIdx = headers.indexOf('postId'); for (let i = rows.length - 1; i >= 1; i--) { if (String(rows[i][pidIdx]) === postId) sheet.deleteRow(i + 1); } } function toggleLike(data) { const sheet = getSheet('posts'); const rows = sheet.getDataRange().getValues(); const headers = rows[0]; const idIdx = headers.indexOf('id'); const likesIdx= headers.indexOf('likes'); for (let i = 1; i < rows.length; i++) { if (String(rows[i][idIdx]) === String(data.postId)) { const current = Number(rows[i][likesIdx]) || 0; const newVal = data.liked ? current + 1 : Math.max(0, current - 1); sheet.getRange(i + 1, likesIdx + 1).setValue(newVal); return { success: true, likes: newVal }; } } return { error: 'Post not found' }; } function adminCheck(pw) { return { success: pw === ADMIN_PW }; }
